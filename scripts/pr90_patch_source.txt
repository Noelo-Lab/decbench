def test_sampleset_exact_key_selects_one_overload_but_keeps_binary_wide_identity(monkeypatch):
    driver = _load_driver()
    from decbench.decompilers.raw import common

    full = {0x1000: "same", 0x2000: "same", 0x3000: "unique"}
    selected = driver._select_manifest_targets(full, {"same@0x2000"})
    assert selected == {0x2000: "same"}
    assert driver._select_manifest_targets(full, {"same"}) == {
        0x1000: "same",
        0x2000: "same",
    }

    result = DecompilationResult(
        binary_path=Path("stripped.bin"),
        binary_name="fixture",
        decompiler=DecompilerMetadata(decompiler_name="fixture"),
        functions={"raw": _fd("sub_2000", 0x2000)},
    )
    monkeypatch.setattr(common, "elf_min_vaddr", lambda _path: 0)
    driver._relabel_to_dwarf(
        result,
        selected,
        Path("unstripped.bin"),
        identity_addr2name=full,
    )

    assert set(result.functions) == {"same@0x2000"}
    assert result.decompiler.failed_functions == []
